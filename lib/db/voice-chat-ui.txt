  <h1>🩺 المساعد الطبي الصوتي</h1>

  <div class="model-selector">
    <button class="model-btn active" onclick="setModel('openai')" id="btn-openai">OpenAI</button>
    <button class="model-btn" onclick="setModel('gemini')" id="btn-gemini">Gemini</button>
  </div>

  <div class="chat-box" id="chatBox">
    <div class="message assistant">
      <div class="model-tag">المساعد الطبي</div>
      مرحباً! أنا مساعدك الطبي. يمكنك التحدث معي بالضغط على زر المايكروفون أو كتابة سؤالك.

  <div class="controls">
    <button class="mic-btn" id="micBtn" onclick="toggleRecording()" title="اضغط للتحدث">🎤</button>
    <div class="status" id="status">اضغط على المايكروفون للتحدث</div>
    <div class="text-input-row">
      <input class="text-input" id="textInput" placeholder="أو اكتب سؤالك هنا..." onkeydown="if(event.key==='Enter') sendText()" />
      <button class="send-btn" onclick="sendText()">إرسال</button>
    </div>
    <div class="action-row">
      <button class="clear-btn" onclick="clearChat()">🗑 مسح المحادثة</button>
      <button class="save-btn" id="saveBtn" onclick="saveHistory()" disabled>💾 حفظ كنص</button>
    </div>
  </div>

  <script>
    let selectedModel = "openai";
    let isRecording = false;
    let conversationHistory = [];
    let recognition = null;

    // الكشف عن دعم SpeechRecognition في المتصفح
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    const hasSpeechAPI = !!SpeechRecognition;

    if (!hasSpeechAPI) {
      document.getElementById("micBtn").title = "متصفحك لا يدعم التعرف على الصوت. استخدم Chrome أو Edge.";
      document.getElementById("micBtn").style.opacity = "0.5";
    }

    function setModel(model) {
      selectedModel = model;
      document.getElementById("btn-openai").classList.toggle("active", model === "openai");
      document.getElementById("btn-gemini").classList.toggle("active", model === "gemini");
    }

    function addMessage(role, text, model = "") {
      const box = document.getElementById("chatBox");
      const div = document.createElement("div");
      div.className = `message ${role}`;
      if (role === "assistant") {
        div.innerHTML = `<div class="model-tag">${model === "gemini" ? "Gemini" : "OpenAI"}</div>${text}`;
      } else {
        div.textContent = text;
      }
      box.appendChild(div);
      box.scrollTop = box.scrollHeight;
    }

    function setStatus(text, loading = false) {
      const el = document.getElementById("status");
      el.textContent = text;
      el.className = loading ? "status loading-dots" : "status";
    }

    async function sendMessage(text) {
      if (!text.trim()) return;

      addMessage("user", text);
      conversationHistory.push({ role: "user", content: text });
      setStatus("جارٍ المعالجة", true);

      try {
        const res = await fetch(`/api/voice/chat/${selectedModel}`, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ message: text, history: conversationHistory.slice(-10) }),
        });

        const data = await res.json();
        if (!res.ok) throw new Error(data.error);

        addMessage("assistant", data.text, selectedModel);
        conversationHistory.push({ role: "assistant", content: data.text, model: selectedModel });
      document.getElementById("saveBtn").disabled = false;

        // تشغيل الصوت
        if (data.audio) {
          const audio = new Audio(data.audio);
          audio.play();
          setStatus("جارٍ التشغيل...");
          audio.onended = () => setStatus("اضغط على المايكروفون للتحدث");
        } else {
          setStatus("اضغط على المايكروفون للتحدث");
        }
      } catch (err) {
        setStatus(`خطأ: ${err.message}`);
      }
    }

    function sendText() {
      const input = document.getElementById("textInput");
      const text = input.value.trim();
      if (!text) return;
      input.value = "";
      sendMessage(text);
    }

    async function saveHistory() {
      if (!conversationHistory.length) return;
      try {
        const res = await fetch("/api/voice/save-history", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ history: conversationHistory }),
        });
        if (!res.ok) throw new Error("فشل الحفظ");
        const blob = await res.blob();
        const url = URL.createObjectURL(blob);
        const a = document.createElement("a");
        a.href = url;
        a.download = `محادثة-طبية-${new Date().toLocaleDateString("ar-SA").replace(/\//g, "-")}.txt`;
        a.click();
        URL.revokeObjectURL(url);
        setStatus("تم حفظ المحادثة بنجاح ✓");
      } catch (err) {
        setStatus(`خطأ في الحفظ: ${err.message}`);
      }
    }

    function clearChat() {
      if (!conversationHistory.length) return;
      if (!confirm("هل تريد مسح المحادثة الحالية؟")) return;
      conversationHistory = [];
      const box = document.getElementById("chatBox");
      box.innerHTML = `<div class="message assistant">
        <div class="model-tag">المساعد الطبي</div>
        مرحباً! أنا مساعدك الطبي. يمكنك التحدث معي بالضغط على زر المايكروفون أو كتابة سؤالك.
      </div>`;
      document.getElementById("saveBtn").disabled = true;
      setStatus("تم مسح المحادثة");
    }

    function toggleRecording() {
      if (!hasSpeechAPI) {
        setStatus("⚠️ متصفحك لا يدعم التعرف على الصوت. استخدم Chrome أو Edge.");
        return;
      }

      if (isRecording) {
        recognition.stop();
        return;
      }

      recognition = new SpeechRecognition();
      recognition.lang = "ar-SA";
      recognition.interimResults = false;
      recognition.maxAlternatives = 1;
      recognition.continuous = false;

      recognition.onstart = () => {
        isRecording = true;
        document.getElementById("micBtn").classList.add("recording");
        document.getElementById("micBtn").textContent = "⏹";
        setStatus("🔴 جارٍ التسجيل... اضغط مرة أخرى للإيقاف");
      };

      recognition.onresult = (event) => {
        const text = event.results[0][0].transcript;
        document.getElementById("textInput").value = text;
        sendMessage(text);
        document.getElementById("textInput").value = "";
      };

      recognition.onerror = (event) => {
        isRecording = false;
        document.getElementById("micBtn").classList.remove("recording");
        document.getElementById("micBtn").textContent = "🎤";

        const errorMessages = {
          "not-allowed": "⚠️ تم رفض إذن المايكروفون. اسمح للمتصفح بالوصول إليه من الإعدادات.",
          "no-speech":   "لم يُكتشف صوت. حاول مرة أخرى.",
          "network":     "خطأ في الشبكة أثناء التعرف على الصوت.",
          "audio-capture": "⚠️ لا يوجد مايكروفون متاح على هذا الجهاز.",
        };
        setStatus(errorMessages[event.error] || `خطأ: ${event.error}`);
      };

      recognition.onend = () => {
        isRecording = false;
        document.getElementById("micBtn").classList.remove("recording");
        document.getElementById("micBtn").textContent = "🎤";
        if (document.getElementById("status").textContent.startsWith("🔴")) {
          setStatus("اضغط على المايكروفون للتحدث");
        }
      };

      try {
        recognition.start();
      } catch (err) {
        setStatus("تعذّر بدء التسجيل: " + err.message);
      }
    }
  </script>
